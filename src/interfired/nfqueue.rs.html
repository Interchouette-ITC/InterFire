4use std::sync::Arc;

6use std::time::Duration;

8use nfq::{Queue, Verdict};
9use tracing::{debug, info, warn};
10
11use crate::packet;
12use crate::shared::Shared;
13
14/// Default NFQUEUE number (matches the isolated NFQUEUE test).
15pub const QUEUE_NUM: u16 = 4242;
16
17/// Bind queue `QUEUE_NUM` and apply pending / default-deny verdicts.
18///
19/// # Errors
20///
21/// Returns I/O errors when the queue cannot be opened or bound.
22pub fn run(shared: &Shared) -> std::io::Result<()> {
23    let mut queue = Queue::open()?;
24    queue.bind(QUEUE_NUM)?;
25    shared.set_enforcement("nfqueue");
26    info!(queue = QUEUE_NUM, "NFQUEUE bound");
27    loop {
28        let mut message = queue.recv()?;
29        let verdict = lookup_verdict(&message, shared);
30        message.set_verdict(verdict);
31        queue.verdict(message)?;
32    }
33}
34
35fn lookup_verdict(message: &nfq::Message, shared: &Shared) -> Verdict {
36    let Some(key) = packet::tcp_destination(message.get_payload()) else {
37        warn!("NFQUEUE packet not parseable as IPv4 TCP; dropping");
38        return Verdict::Drop;
39    };
40    // Observation may still be draining the ring buffer after `tcp_v4_connect`;
41    // the packet is held in NFQUEUE, so a short wait stays fail-closed.
42    for attempt in 0..50 {
43        let Ok(mut pending) = shared.pending.lock() else {
44            return Verdict::Drop;
45        };
46        if let Some(verdict) = pending.take(key) {
47            debug!(
48                port = key.port,
49                attempt,
50                ?verdict,
51                "NFQUEUE matched pending decision"
52            );
53            return verdict;
54        }
55        drop(pending);
56        thread::sleep(Duration::from_millis(1));
57    }
58    warn!(
59        port = key.port,
60        "NFQUEUE without pending decision; dropping"
61    );
62    Verdict::Drop
63}
64
65/// Attempt to bind; on failure mark enforcement degraded and return.
66pub fn run_or_degrade(shared: &Arc<Shared>) {
67    if let Err(error) = run(shared) {
68        shared.set_enforcement("degraded");
69        warn!(%error, "NFQUEUE unavailable; enforcement=degraded");
70    }
71}
