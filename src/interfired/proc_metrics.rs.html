1//! Lightweight `/proc/self` RSS and CPU jiffy samples for status metrics.
2#![forbid(unsafe_code)]
3
4use std::fs;
5use std::io;
6#[cfg(test)]
7use std::sync::atomic::{AtomicBool, Ordering};
8
9/// Snapshot of this process for IPC / UI profiling.
10#[derive(Clone, Copy, Debug, Eq, PartialEq)]
11pub struct SelfMetrics {
12    pub pid: u32,
13    pub rss_kib: u64,
14    /// `utime + stime` from `/proc/self/stat` (clock ticks).
15    pub cpu_jiffies: u64,
16}
17
18#[cfg(test)]
19static FORCE_SAMPLE_FAILURE: AtomicBool = AtomicBool::new(false);
20#[cfg(test)]
21static SAMPLE_TEST_LOCK: std::sync::Mutex<()> = std::sync::Mutex::new(());
22
23/// Read pid, `VmRSS`, and CPU jiffies for the current process.
24///
25/// # Errors
26///
27/// Returns I/O or parse failures when `/proc/self` is unreadable.
28pub fn sample_self() -> io::Result<SelfMetrics> {
29    #[cfg(test)]
30    if FORCE_SAMPLE_FAILURE.load(Ordering::Relaxed) {
31        return Err(io::Error::other("forced sample_self failure"));
32    }
33    let pid = std::process::id();
34    let rss_kib = read_vm_rss_kib()?;
35    let cpu_jiffies = read_cpu_jiffies()?;
36    Ok(SelfMetrics {
37        pid,
38        rss_kib,
39        cpu_jiffies,
40    })
41}
42
43/// Force [`sample_self`] to fail for the lifetime of the guard (unit tests only).
44#[cfg(test)]
45pub struct ForceSampleFailure {
46    _guard: std::sync::MutexGuard<'static, ()>,
47}
48
49#[cfg(test)]
50impl ForceSampleFailure {
51    #[must_use]
52    pub fn arm() -> Self {
53        let guard = SAMPLE_TEST_LOCK.lock().expect("sample test lock");
54        FORCE_SAMPLE_FAILURE.store(true, Ordering::Relaxed);
55        Self { _guard: guard }
56    }
57}
58
59#[cfg(test)]
60impl Drop for ForceSampleFailure {
61    fn drop(&mut self) {
62        FORCE_SAMPLE_FAILURE.store(false, Ordering::Relaxed);
63    }
64}

66fn parse_vm_rss_kib(status: &str) -> io::Result<u64> {
67    for line in status.lines() {
68        if let Some(rest) = line.strip_prefix("VmRSS:") {
69            let kib = rest
70                .split_whitespace()
71                .next()
72                .ok_or_else(|| io::Error::new(io::ErrorKind::InvalidData, "VmRSS missing value"))?;
73            return kib.parse().map_err(|error| {
74                io::Error::new(io::ErrorKind::InvalidData, format!("VmRSS: {error}"))
75            });
76        }
77    }
78    Err(io::Error::new(io::ErrorKind::InvalidData, "VmRSS missing"))
79}
80
81fn read_vm_rss_kib() -> io::Result<u64> {
82    parse_vm_rss_kib(&fs::read_to_string("/proc/self/status")?)
83}
84
85fn parse_cpu_jiffies(stat: &str) -> io::Result<u64> {
86    // Field 1 can contain spaces inside parentheses; split after the last ')'.
87    let after_comm = stat
88        .rsplit_once(')')
89        .map(|(_, rest)| rest.trim_start())
90        .ok_or_else(|| io::Error::new(io::ErrorKind::InvalidData, "stat missing comm"))?;
91    let mut fields = after_comm.split_whitespace();
92    // After comm: state(2) … utime(14) stime(15) relative to full stat = indices 11,12 of remainder.
93    let utime = fields
94        .nth(11)
95        .ok_or_else(|| io::Error::new(io::ErrorKind::InvalidData, "stat missing utime"))?;
96    let stime = fields
97        .next()
98        .ok_or_else(|| io::Error::new(io::ErrorKind::InvalidData, "stat missing stime"))?;
99    let utime: u64 = utime
100        .parse()
101        .map_err(|error| io::Error::new(io::ErrorKind::InvalidData, format!("utime: {error}")))?;
102    let stime: u64 = stime
103        .parse()
104        .map_err(|error| io::Error::new(io::ErrorKind::InvalidData, format!("stime: {error}")))?;
105    Ok(utime.saturating_add(stime))
106}
107
108fn read_cpu_jiffies() -> io::Result<u64> {
109    parse_cpu_jiffies(&fs::read_to_string("/proc/self/stat")?)
110}

112#[cfg(test)]
113mod tests {
114    use super::*;
115
116    #[test]
117    fn sample_self_reads_positive_rss() {
118        let metrics = sample_self().expect("/proc/self");
119        assert_eq!(metrics.pid, std::process::id());
120        assert!(metrics.rss_kib > 0);
121    }
122
123    #[test]
124    fn parse_vm_rss_handles_valid_and_invalid_status() {
125        assert_eq!(parse_vm_rss_kib("VmRSS:\t2048 kB\n").unwrap(), 2048);
126        assert!(parse_vm_rss_kib("Name:\tbash\n").is_err());
127        assert!(parse_vm_rss_kib("VmRSS:\tbad\n").is_err());
128    }
129
130    #[test]
131    fn parse_cpu_jiffies_handles_valid_and_invalid_stat() {
132        let mut fields = vec!["S".to_owned()];
133        fields.extend((0..10).map(|_| "0".to_owned()));
134        fields.push("100".to_owned());
135        fields.push("200".to_owned());
136        let stat = format!("42 (worker) {}", fields.join(" "));
137        assert_eq!(parse_cpu_jiffies(&stat).unwrap(), 300);
138        assert!(parse_cpu_jiffies("broken").is_err());
139        assert!(parse_cpu_jiffies("1 (a) S").is_err());
140    }
