1//! Bounded recent outbound destinations per process identity.
2#![forbid(unsafe_code)]
3
4use std::collections::{HashMap, VecDeque};
5use std::net::Ipv4Addr;
6
7/// One recent connect target for an observed process.
8#[derive(Clone, Debug, Eq, PartialEq)]
9pub struct RecentDest {
10    pub ipv4: Ipv4Addr,
11    pub port: u16,
12    pub verdict: String,
13}
14
15/// Per-identity ring of recent destinations (firewall-observed only).
16pub struct RecentConnects {
17    capacity_keys: usize,
18    per_key: usize,
19    order: VecDeque<(u32, u64)>,
20    entries: HashMap<(u32, u64), VecDeque<RecentDest>>,
21}
22
23impl RecentConnects {
24    /// Create a bounded recent-connect table.
25    ///
26    /// # Panics
27    ///
28    /// Panics when either capacity is zero.
29    #[must_use]
30    pub fn new(capacity_keys: usize, per_key: usize) -> Self {
31        assert!(capacity_keys > 0 && per_key > 0);
32        Self {
33            capacity_keys,
34            per_key,
35            order: VecDeque::new(),
36            entries: HashMap::new(),
37        }
38    }
39
40    /// Record a destination for `(pid, start_ticks)`, newest last.
41    pub fn record(&mut self, pid: u32, start_ticks: u64, dest: RecentDest) {
42        let key = (pid, start_ticks);
43        if !self.entries.contains_key(&key) {
44            while self.entries.len() >= self.capacity_keys {
45                if let Some(oldest) = self.order.pop_front() {
46                    self.entries.remove(&oldest);
47                } else {
48                    break;
49                }
50            }
51            self.order.push_back(key);
52            self.entries.insert(key, VecDeque::new());
53        }
54        let queue = self.entries.get_mut(&key).expect("just inserted");
55        if let Some(existing) = queue
56            .iter_mut()
57            .find(|row| row.ipv4 == dest.ipv4 && row.port == dest.port)
58        {
59            existing.verdict.clone_from(&dest.verdict);
60            return;
61        }
62        while queue.len() >= self.per_key {
63            queue.pop_front();
64        }
65        queue.push_back(dest);
66    }
67
68    /// Destinations for one identity (oldest first).
69    #[must_use]
70    pub fn for_process(&self, pid: u32, start_ticks: u64) -> Vec<RecentDest> {
71        self.entries
72            .get(&(pid, start_ticks))
73            .map(|queue| queue.iter().cloned().collect())
74            .unwrap_or_default()
75    }
76}

78#[cfg(test)]
79mod tests {
80    use super::*;
81
82    #[test]
83    fn caps_per_key_and_updates_verdict() {
84        let mut recent = RecentConnects::new(2, 2);
85        let dest = |port, verdict: &str| RecentDest {
86            ipv4: Ipv4Addr::new(203, 0, 113, 10),
87            port,
88            verdict: verdict.into(),
89        };
90        recent.record(1, 9, dest(443, "prompt"));
91        recent.record(1, 9, dest(80, "allow"));
92        recent.record(1, 9, dest(22, "deny"));
93        let rows = recent.for_process(1, 9);
94        assert_eq!(rows.len(), 2);
95        assert_eq!(rows[0].port, 80);
96        assert_eq!(rows[1].port, 22);
97        recent.record(1, 9, dest(80, "deny"));
98        let rows = recent.for_process(1, 9);
99        assert_eq!(rows[0].verdict, "deny");
100    }
