19const ENFORCEMENT_NONE: u8 = 0;
20const ENFORCEMENT_NFQUEUE: u8 = 1;
21const ENFORCEMENT_DEGRADED: u8 = 2;
22
23const OBSERVATION_ATTACHED: u8 = 0;
24const OBSERVATION_DEGRADED: u8 = 1;
25
26/// Cross-thread daemon state.
27pub struct Shared {
28    pub rules: Mutex<RuleSet>,
29    pub store: RulesStore,
30    pub pending: Mutex<PendingTable>,
31    pub process_cache: Mutex<ProcessCache>,
32    pub recent: Mutex<RecentConnects>,
33    pub prompts: Mutex<PromptQueue>,
34    pub dns: Mutex<DnsCache>,
35    pub audit: Mutex<AuditLog>,
36    enforcement: AtomicU8,
37    observation: AtomicU8,
38}
39
40impl Shared {
41    /// Build shared state.
42    ///
43    /// # Errors
44    ///
45    /// Returns I/O failures while opening the audit log.
46    pub fn new(
47        rules: RuleSet,
48        store: RulesStore,
49        observation: &'static str,
50        pending_capacity: usize,
51        pending_ttl: Duration,
52        process_capacity: usize,
53        audit_path: PathBuf,
54    ) -> std::io::Result<Self> {
55        Ok(Self {
56            rules: Mutex::new(rules),
57            store,
58            pending: Mutex::new(PendingTable::new(pending_capacity, pending_ttl)),
59            process_cache: Mutex::new(ProcessCache::new(process_capacity)),
60            recent: Mutex::new(RecentConnects::new(process_capacity, 8)),
61            prompts: Mutex::new(PromptQueue::with_defaults()),
62            dns: Mutex::new(DnsCache::with_defaults()),
63            audit: Mutex::new(AuditLog::open(
64                audit_path,
65                DEFAULT_AUDIT_MAX_BYTES,
66                MAX_LOG_RECORDS_PER_SUBSCRIBER,
67            )?),
68            enforcement: AtomicU8::new(ENFORCEMENT_NONE),
69            observation: AtomicU8::new(observation_code(observation)),
70        })
71    }
72
73    pub fn set_enforcement(&self, label: &'static str) {
74        self.enforcement
75            .store(enforcement_code(label), Ordering::Relaxed);
76    }
77
78    #[must_use]
79    pub fn enforcement(&self) -> &'static str {
80        enforcement_label(self.enforcement.load(Ordering::Relaxed))
81    }
82
83    #[must_use]
84    pub fn observation(&self) -> &'static str {
85        observation_label(self.observation.load(Ordering::Relaxed))
86    }
87
88    /// Replace the prompt queue (unit tests only).
89    #[cfg(test)]
90    pub fn set_prompt_queue(&self, queue: PromptQueue) {
91        *self.prompts.lock().expect("prompts lock") = queue;
92    }
93}
94
95fn enforcement_code(label: &str) -> u8 {
96    match label {
97        "nfqueue" => ENFORCEMENT_NFQUEUE,
98        "degraded" => ENFORCEMENT_DEGRADED,
99        _ => ENFORCEMENT_NONE,
100    }
101}
102
103const fn enforcement_label(code: u8) -> &'static str {
104    match code {
105        ENFORCEMENT_NFQUEUE => "nfqueue",
106        ENFORCEMENT_DEGRADED => "degraded",
107        _ => "none",
108    }
109}
110
111fn observation_code(label: &str) -> u8 {
112    if label == "attached" {
113        OBSERVATION_ATTACHED
114    } else {
115        OBSERVATION_DEGRADED
116    }
117}
118
119const fn observation_label(code: u8) -> &'static str {
120    if code == OBSERVATION_ATTACHED {
121        "attached"
122    } else {
123        "degraded"
124    }
125}

127#[cfg(test)]
128mod tests {
129    use std::fs;
130    use std::time::Duration;
131
132    use interfire_rules::RulesStore;
133
134    use super::*;
135
136    fn temp_audit(name: &str) -> PathBuf {
137        std::env::temp_dir().join(format!("interfire-shared-{}-{name}", std::process::id()))
138    }
139
140    #[test]
141    fn new_initializes_state_and_audit_log() {
142        let audit_path = temp_audit("new.log");
143        let _ = fs::remove_file(&audit_path);
144        let store = RulesStore::new(audit_path.with_extension("rules.toml"));
145        let shared = Shared::new(
146            RuleSet::default(),
147            store,
148            "attached",
149            8,
150            Duration::from_secs(5),
151            8,
152            audit_path.clone(),
153        )
154        .expect("shared state");
155        assert_eq!(shared.observation(), "attached");
156        assert_eq!(shared.enforcement(), "none");
157        shared.audit.lock().unwrap().append("boot");
158        assert!(audit_path.exists());
159        let _ = fs::remove_file(audit_path);
160    }
161
162    #[test]
163    fn new_fails_when_audit_path_is_a_directory() {
164        let audit_dir = temp_audit("dir");
165        let _ = fs::remove_dir_all(&audit_dir);
166        fs::create_dir_all(&audit_dir).expect("mkdir");
167        let store = RulesStore::new(audit_dir.join("rules.toml"));
168        let result = Shared::new(
169            RuleSet::default(),
170            store,
171            "attached",
172            8,
173            Duration::from_secs(5),
174            8,
175            audit_dir,
176        );
177        assert!(result.is_err());
178    }
179
180    #[test]
181    fn enforcement_setters_and_getters() {
182        let audit_path = temp_audit("enforce.log");
183        let _ = fs::remove_file(&audit_path);
184        let store = RulesStore::new(audit_path.with_extension("rules.toml"));
185        let shared = Shared::new(
186            RuleSet::default(),
187            store,
188            "degraded",
189            8,
190            Duration::from_secs(5),
191            8,
192            audit_path.clone(),
193        )
194        .expect("shared state");
195        assert_eq!(shared.observation(), "degraded");
196        shared.set_enforcement("nfqueue");
197        assert_eq!(shared.enforcement(), "nfqueue");
198        shared.set_enforcement("degraded");
199        assert_eq!(shared.enforcement(), "degraded");
200        shared.set_enforcement("unknown");
201        assert_eq!(shared.enforcement(), "none");
202        let _ = fs::remove_file(audit_path);
203    }
