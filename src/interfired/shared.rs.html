1//! Shared daemon state for IPC, observation, and NFQUEUE threads.
2#![forbid(unsafe_code)]
3
4use std::sync::Mutex;
5use std::sync::atomic::{AtomicU8, Ordering};
6use std::time::Duration;
7
8use interfire_rules::{RuleSet, RulesStore};
9
10use crate::pending::PendingTable;
11use crate::process::ProcessCache;
12use crate::prompts::PromptQueue;
13
14const ENFORCEMENT_NONE: u8 = 0;
15const ENFORCEMENT_NFQUEUE: u8 = 1;
16const ENFORCEMENT_DEGRADED: u8 = 2;
17
18const OBSERVATION_ATTACHED: u8 = 0;
19const OBSERVATION_DEGRADED: u8 = 1;
20
21/// Cross-thread daemon state.
22pub struct Shared {
23    pub rules: Mutex<RuleSet>,
24    pub store: RulesStore,
25    pub pending: Mutex<PendingTable>,
26    pub process_cache: Mutex<ProcessCache>,
27    pub prompts: Mutex<PromptQueue>,
28    enforcement: AtomicU8,
29    observation: AtomicU8,
30}
31
32impl Shared {
33    #[must_use]
34    pub fn new(
35        rules: RuleSet,
36        store: RulesStore,
37        observation: &'static str,
38        pending_capacity: usize,
39        pending_ttl: Duration,
40        process_capacity: usize,
41    ) -> Self {
42        Self {
43            rules: Mutex::new(rules),
44            store,
45            pending: Mutex::new(PendingTable::new(pending_capacity, pending_ttl)),
46            process_cache: Mutex::new(ProcessCache::new(process_capacity)),
47            prompts: Mutex::new(PromptQueue::with_defaults()),
48            enforcement: AtomicU8::new(ENFORCEMENT_NONE),
49            observation: AtomicU8::new(observation_code(observation)),
50        }
51    }
52
53    pub fn set_enforcement(&self, label: &'static str) {
54        self.enforcement
55            .store(enforcement_code(label), Ordering::Relaxed);
56    }
57
58    #[must_use]
59    pub fn enforcement(&self) -> &'static str {
60        enforcement_label(self.enforcement.load(Ordering::Relaxed))
61    }
62
63    #[must_use]
64    pub fn observation(&self) -> &'static str {
65        observation_label(self.observation.load(Ordering::Relaxed))
66    }
67}
68
69fn enforcement_code(label: &str) -> u8 {
70    match label {
71        "nfqueue" => ENFORCEMENT_NFQUEUE,
72        "degraded" => ENFORCEMENT_DEGRADED,
73        _ => ENFORCEMENT_NONE,
74    }
75}
76
77const fn enforcement_label(code: u8) -> &'static str {
78    match code {
79        ENFORCEMENT_NFQUEUE => "nfqueue",
80        ENFORCEMENT_DEGRADED => "degraded",
81        _ => "none",
82    }
83}
84
85fn observation_code(label: &str) -> u8 {
86    if label == "attached" {
87        OBSERVATION_ATTACHED
88    } else {
89        OBSERVATION_DEGRADED
90    }
91}
92
93const fn observation_label(code: u8) -> &'static str {
94    if code == OBSERVATION_ATTACHED {
95        "attached"
96    } else {
97        "degraded"
98    }
99}
