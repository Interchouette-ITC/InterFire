67    AuditSubscribe {

71    ProcessList,

74impl Request {
75    /// Parse one newline-terminated `v1 …` control frame.
76    ///
77    /// # Errors
78    ///
79    /// Returns [`ProtocolError::UnsupportedVersion`] when the frame is not `v1`,
80    /// or [`ProtocolError::Malformed`] when the command or fields are invalid.
81    pub fn parse(frame: &str) -> Result<Self, ProtocolError> {
82        let mut fields = frame.split_whitespace();
83        let version = fields.next().ok_or(ProtocolError::Malformed)?;
84        if version != "v1" {
85            return Err(ProtocolError::UnsupportedVersion);
86        }
87        match fields.next() {
88            Some("ping") if fields.next().is_none() => Ok(Self::Ping),
89            Some("status") if fields.next().is_none() => Ok(Self::Status),
90            Some("rule-list") if fields.next().is_none() => Ok(Self::RuleList),
91            Some("rule-delete") => {
92                let id = fields.next().and_then(|value| value.parse().ok());
93                if fields.next().is_none() {
94                    id.map(|id| Self::RuleDelete { id })
95                        .ok_or(ProtocolError::Malformed)
96                } else {
97                    Err(ProtocolError::Malformed)
98                }
99            }
100            Some("rule-add") => {
101                let id = fields.next().and_then(|value| value.parse().ok());
102                let executable = fields.next().map(str::to_owned);
103                let verdict = fields.next().map(str::to_owned);
104                let port = fields.next().and_then(|value| value.parse().ok());
105                if fields.next().is_some() {
106                    return Err(ProtocolError::Malformed);
107                }
108                match (id, executable, verdict, port) {
109                    (Some(id), Some(executable), Some(verdict), Some(port)) => Ok(Self::RuleAdd {
110                        id,
111                        executable,
112                        verdict,
113                        port,
114                    }),
115                    _ => Err(ProtocolError::Malformed),
116                }
117            }
118            Some("prompt-list") if fields.next().is_none() => Ok(Self::PromptList),
119            Some("prompt-answer") => {
120                let id = fields.next().and_then(|value| value.parse().ok());
121                let verdict = fields.next().map(str::to_owned);
122                let scope = fields.next().map(str::to_owned);
123                if fields.next().is_some() {
124                    return Err(ProtocolError::Malformed);
125                }
126                match (id, verdict, scope) {
127                    (Some(id), Some(verdict), Some(scope)) => {
128                        Ok(Self::PromptAnswer { id, verdict, scope })
129                    }
130                    _ => Err(ProtocolError::Malformed),
131                }
132            }
133            Some("dns-list") if fields.next().is_none() => Ok(Self::DnsList),
134            Some("dns-note") => {
135                let hostname = fields.next().map(str::to_owned);
136                let ipv4 = fields.next().map(str::to_owned);
137                let ttl_secs = match fields.next() {
138                    Some(value) => Some(value.parse().map_err(|_| ProtocolError::Malformed)?),
139                    None => None,
140                };
141                if fields.next().is_some() {
142                    return Err(ProtocolError::Malformed);
143                }
144                match (hostname, ipv4) {
145                    (Some(hostname), Some(ipv4)) => Ok(Self::DnsNote {
146                        hostname,
147                        ipv4,
148                        ttl_secs,
149                    }),
150                    _ => Err(ProtocolError::Malformed),
151                }
152            }
153            Some("audit-tail") => {
154                let limit = match fields.next() {
155                    Some(value) => value.parse().map_err(|_| ProtocolError::Malformed)?,
156                    None => MAX_LOG_RECORDS_PER_SUBSCRIBER,
157                };
158                if fields.next().is_some() {
159                    return Err(ProtocolError::Malformed);
160                }
161                Ok(Self::AuditTail { limit })
162            }
163            Some("audit-subscribe") => {
164                let id = fields.next().map(str::to_owned);
165                let since = match fields.next() {
166                    Some(value) => value
167                        .strip_prefix("since=")
168                        .and_then(|value| value.parse().ok())
169                        .ok_or(ProtocolError::Malformed)?,
170                    None => 0,
171                };
172                if fields.next().is_some() {
173                    return Err(ProtocolError::Malformed);
174                }
175                id.map(|id| Self::AuditSubscribe { id, since })
176                    .ok_or(ProtocolError::Malformed)
177            }
178            Some("process-list") if fields.next().is_none() => Ok(Self::ProcessList),
179            _ => Err(ProtocolError::Malformed),
180        }
181    }
182}

197/// Daemon `status` response body (encode side).
198#[derive(Clone, Copy, Debug, Eq, PartialEq)]
199pub struct StatusBody {
200    /// Verdict path state (`none`, `nfqueue`, or `degraded`).
201    pub enforcement: &'static str,
202    /// eBPF observation state (`attached` or `degraded`).
203    pub observation: &'static str,
204    pub ipc_version: u16,
205    /// Daemon process id.
206    pub pid: u32,
207    /// Daemon `VmRSS` in KiB from `/proc/self/status`.
208    pub rss_kib: u64,
209    /// Daemon `utime + stime` jiffies from `/proc/self/stat` (for CPU % over polls).
210    pub cpu_jiffies: u64,
211}
212
213impl Response {
214    /// Encode a response as one newline-terminated control frame.
215    #[must_use]
216    pub fn encode(&self) -> String {
217        match self {
218            Self::Pong => "v1 pong\n".into(),
219            Self::Status(body) => format!(
220                "v1 status enforcement={} observation={} ipc_version={} pid={} rss_kib={} cpu_jiffies={}\n",
221                body.enforcement,
222                body.observation,
223                body.ipc_version,
224                body.pid,
225                body.rss_kib,
226                body.cpu_jiffies,
227            ),
228            Self::Error(message) => format!("v1 error {message}\n"),
229            Self::Rules(value) => format!("v1 rules {value}\n"),
230            Self::Prompts(value) => format!("v1 prompts {value}\n"),
231            Self::Dns(value) => format!("v1 dns {value}\n"),
232            Self::Audit(value) => format!("v1 audit-tail {value}\n"),
233            Self::Subscribed(id) => format!("v1 subscribed {id}\n"),
234            Self::Processes(value) => format!("v1 processes {value}\n"),
235        }
236    }
237}

239/// Parsed daemon status fields (owned; for clients).
240#[derive(Clone, Debug, Eq, PartialEq)]
241pub struct DaemonStatus {
242    pub enforcement: String,
243    pub observation: String,
244    pub ipc_version: u16,
245    /// Present when the daemon includes process metrics on `status`.
246    pub pid: Option<u32>,
247    /// Daemon `VmRSS` in KiB when present.
248    pub rss_kib: Option<u64>,
249    /// Daemon CPU jiffies (`utime + stime`) when present.
250    pub cpu_jiffies: Option<u64>,
251}
252
253impl DaemonStatus {
254    /// Parse a `v1 status …` response frame.
255    ///
256    /// Unknown `key=value` fields are ignored so older clients keep working when
257    /// the daemon adds metrics keys.
258    ///
259    /// # Errors
260    ///
261    /// Returns [`ProtocolError`] when the frame is not a well-formed status line.
262    pub fn parse(frame: &str) -> Result<Self, ProtocolError> {
263        let line = frame.trim_end_matches(['\r', '\n']);
264        let mut fields = line.split_whitespace();
265        match fields.next() {
266            Some("v1") => {}
267            Some(token) if token.starts_with('v') => return Err(ProtocolError::UnsupportedVersion),
268            _ => return Err(ProtocolError::Malformed),
269        }
270        if fields.next() != Some("status") {
271            return Err(ProtocolError::Malformed);
272        }
273        let mut enforcement = None;
274        let mut observation = None;
275        let mut ipc_version = None;
276        let mut pid = None;
277        let mut rss_kib = None;
278        let mut cpu_jiffies = None;
279        for field in fields {
280            if let Some(value) = field.strip_prefix("enforcement=") {
281                enforcement = Some(value.to_owned());
282            } else if let Some(value) = field.strip_prefix("observation=") {
283                observation = Some(value.to_owned());
284            } else if let Some(value) = field.strip_prefix("ipc_version=") {
285                ipc_version = Some(value.parse().map_err(|_| ProtocolError::Malformed)?);
286            } else if let Some(value) = field.strip_prefix("pid=") {
287                pid = Some(value.parse().map_err(|_| ProtocolError::Malformed)?);
288            } else if let Some(value) = field.strip_prefix("rss_kib=") {
289                rss_kib = Some(value.parse().map_err(|_| ProtocolError::Malformed)?);
290            } else if let Some(value) = field.strip_prefix("cpu_jiffies=") {
291                cpu_jiffies = Some(value.parse().map_err(|_| ProtocolError::Malformed)?);
292            } else if field.contains('=') {
293                // Forward-compatible: ignore unknown keys.
294            } else {
295                return Err(ProtocolError::Malformed);
296            }
297        }
298        Ok(Self {
299            enforcement: enforcement.ok_or(ProtocolError::Malformed)?,
300            observation: observation.ok_or(ProtocolError::Malformed)?,
301            ipc_version: ipc_version.ok_or(ProtocolError::Malformed)?,
302            pid,
303            rss_kib,
304            cpu_jiffies,
305        })
306    }
307}

309/// One streamed audit record (`v1 audit SEQ|message`).
310#[derive(Clone, Debug, Eq, PartialEq)]
311pub struct AuditStreamRecord {
312    pub sequence: u64,
313    pub message: String,
314}
315
316impl AuditStreamRecord {
317    /// Parse a streamed audit frame.
318    ///
319    /// # Errors
320    ///
321    /// Returns [`ProtocolError`] when the frame is not `v1 audit SEQ|message`.
322    pub fn parse(frame: &str) -> Result<Self, ProtocolError> {
323        let line = frame.trim_end_matches(['\r', '\n']);
324        let mut fields = line.splitn(3, ' ');
325        match fields.next() {
326            Some("v1") => {}
327            Some(token) if token.starts_with('v') => return Err(ProtocolError::UnsupportedVersion),
328            _ => return Err(ProtocolError::Malformed),
329        }
330        if fields.next() != Some("audit") {
331            return Err(ProtocolError::Malformed);
332        }
333        let payload = fields.next().ok_or(ProtocolError::Malformed)?;
334        let (sequence, message) = payload.split_once('|').ok_or(ProtocolError::Malformed)?;
335        Ok(Self {
336            sequence: sequence.parse().map_err(|_| ProtocolError::Malformed)?,
337            message: message.to_owned(),
338        })
339    }
340}

342/// One rule row as returned by `v1 rules …`.
343#[derive(Clone, Debug, Eq, PartialEq)]

345    pub id: u64,
346    pub executable: String,
347    pub verdict: String,
348    pub port: u16,
349}
350
351impl RuleRow {
352    /// Parse a `v1 rules …` response frame into rows.
353    ///
354    /// # Errors
355    ///
356    /// Returns [`ProtocolError`] when the frame is not a well-formed rules list.
357    pub fn parse_frame(frame: &str) -> Result<Vec<Self>, ProtocolError> {
358        let line = frame.trim_end_matches(['\r', '\n']);
359        let mut fields = line.splitn(3, ' ');
360        match fields.next() {
361            Some("v1") => {}
362            Some(token) if token.starts_with('v') => return Err(ProtocolError::UnsupportedVersion),
363            _ => return Err(ProtocolError::Malformed),
364        }
365        if fields.next() != Some("rules") {
366            return Err(ProtocolError::Malformed);
367        }
368        let payload = fields.next().unwrap_or("");
369        if payload.is_empty() {
370            return Ok(Vec::new());
371        }
372        payload.split(',').map(Self::parse_row).collect()
373    }
374
375    fn parse_row(row: &str) -> Result<Self, ProtocolError> {
376        let mut parts = row.splitn(4, '|');
377        let id = parts
378            .next()
379            .and_then(|value| value.parse().ok())
380            .ok_or(ProtocolError::Malformed)?;
381        let executable = parts.next().ok_or(ProtocolError::Malformed)?.to_owned();
382        let verdict = parts.next().ok_or(ProtocolError::Malformed)?.to_owned();
383        let port = parts
384            .next()
385            .and_then(|value| value.parse().ok())
386            .ok_or(ProtocolError::Malformed)?;
387        Ok(Self {
388            id,
389            executable,
390            verdict,
391            port,
392        })
393    }
394
395    /// Compact list label: `id exe verdict port`.
396    #[must_use]
397    pub fn list_label(&self) -> String {
398        format!(
399            "{}  {}  {}  :{}",
400            self.id, self.executable, self.verdict, self.port
401        )
402    }

405/// One pending prompt row as returned by `v1 prompts …`.
406#[derive(Clone, Debug, Eq, PartialEq)]
407pub struct PromptRow {
408    pub id: u64,
409    pub executable: String,
410    pub destination: String,
411    pub port: u16,
412    pub protocol: String,
413    pub remaining_secs: u64,
414}
415
416impl PromptRow {
417    /// Parse a `v1 prompts …` response frame into rows.
418    ///
419    /// # Errors
420    ///
421    /// Returns [`ProtocolError`] when the frame is not a well-formed prompts list.
422    pub fn parse_frame(frame: &str) -> Result<Vec<Self>, ProtocolError> {
423        let line = frame.trim_end_matches(['\r', '\n']);
424        let mut fields = line.splitn(3, ' ');
425        match fields.next() {
426            Some("v1") => {}
427            Some(token) if token.starts_with('v') => return Err(ProtocolError::UnsupportedVersion),
428            _ => return Err(ProtocolError::Malformed),
429        }
430        if fields.next() != Some("prompts") {
431            return Err(ProtocolError::Malformed);
432        }
433        let payload = fields.next().unwrap_or("");
434        if payload.is_empty() {
435            return Ok(Vec::new());
436        }
437        payload.split(',').map(Self::parse_row).collect()
438    }
439
440    fn parse_row(row: &str) -> Result<Self, ProtocolError> {
441        let parts: Vec<&str> = row.split('|').collect();
442        if parts.len() < 4 {
443            return Err(ProtocolError::Malformed);
444        }
445        let id = parts[0].parse().map_err(|_| ProtocolError::Malformed)?;
446        let executable = parts[1].to_owned();
447        let destination = parts[2].to_owned();
448        let port = parts[3].parse().map_err(|_| ProtocolError::Malformed)?;
449        let protocol = parts.get(4).copied().unwrap_or("tcp").to_owned();
450        let remaining_secs = parts
451            .get(5)
452            .and_then(|value| value.parse().ok())
453            .unwrap_or(0);
454        Ok(Self {
455            id,
456            executable,
457            destination,
458            port,
459            protocol,
460            remaining_secs,
461        })
462    }
463
464    /// Compact list label.
465    #[must_use]
466    pub fn list_label(&self) -> String {
467        format!(
468            "{}  {}  {}:{} {}  {}s",
469            self.id,
470            self.executable,
471            self.destination,
472            self.port,
473            self.protocol,
474            self.remaining_secs
475        )
476    }
477
478    /// Whether the TUI should allow answering this prompt.
479    #[must_use]
480    pub const fn can_answer(&self) -> bool {
481        self.remaining_secs > 0
482    }
483}

485/// One observed process row as returned by `v1 processes …`.
486#[derive(Clone, Debug, Eq, PartialEq)]
487pub struct ProcessRow {
488    pub pid: u32,
489    pub start_ticks: u64,
490    pub uid: u32,
491    pub executable: String,
492    pub cmdline: String,
493    /// Effective rule verdict for this executable (`allow` / `deny` / `prompt`).
494    pub verdict: String,
495    /// Compact recent destinations: `ip:port/verdict+…`.
496    pub ports: String,
497}
498
499impl ProcessRow {
500    /// Parse a `v1 processes …` response frame into rows.
501    ///
502    /// # Errors
503    ///
504    /// Returns [`ProtocolError`] when the frame is not a well-formed process list.
505    pub fn parse_frame(frame: &str) -> Result<Vec<Self>, ProtocolError> {
506        let line = frame.trim_end_matches(['\r', '\n']);
507        let mut fields = line.splitn(3, ' ');
508        match fields.next() {
509            Some("v1") => {}
510            Some(token) if token.starts_with('v') => return Err(ProtocolError::UnsupportedVersion),
511            _ => return Err(ProtocolError::Malformed),
512        }
513        if fields.next() != Some("processes") {
514            return Err(ProtocolError::Malformed);
515        }
516        let payload = fields.next().unwrap_or("");
517        if payload.is_empty() {
518            return Ok(Vec::new());
519        }
520        payload.split(';').map(Self::parse_row).collect()
521    }
522
523    fn parse_row(row: &str) -> Result<Self, ProtocolError> {
524        let mut parts = row.splitn(7, '|');
525        let pid = parts
526            .next()
527            .and_then(|value| value.parse().ok())
528            .ok_or(ProtocolError::Malformed)?;
529        let start_ticks = parts
530            .next()
531            .and_then(|value| value.parse().ok())
532            .ok_or(ProtocolError::Malformed)?;
533        let uid = parts
534            .next()
535            .and_then(|value| value.parse().ok())
536            .ok_or(ProtocolError::Malformed)?;
537        let verdict = parts.next().ok_or(ProtocolError::Malformed)?.to_owned();
538        let recent_ports = parts.next().ok_or(ProtocolError::Malformed)?.to_owned();
539        let executable = unescape_field(parts.next().ok_or(ProtocolError::Malformed)?);
540        let cmdline = unescape_field(parts.next().ok_or(ProtocolError::Malformed)?);
541        Ok(Self {
542            pid,
543            start_ticks,
544            uid,
545            executable,
546            cmdline,
547            verdict,
548            ports: recent_ports,
549        })
550    }
551
552    /// Encode one row for a `v1 processes` payload.
553    #[must_use]
554    pub fn encode_row(&self) -> String {
555        format!(
556            "{}|{}|{}|{}|{}|{}|{}",
557            self.pid,
558            self.start_ticks,
559            self.uid,
560            self.verdict,
561            self.ports,
562            escape_field(&self.executable),
563            escape_field(&self.cmdline),
564        )
565    }
566
567    /// Compact list label for UI / TUI.
568    #[must_use]
569    pub fn list_label(&self) -> String {
570        format!(
571            "{}  {}  {}  {}",
572            self.pid, self.executable, self.verdict, self.ports
573        )
574    }
575}
576
577fn escape_field(value: &str) -> String {
578    value
579        .replace('%', "%25")
580        .replace('|', "%7C")
581        .replace(';', "%3B")
582}
583
584fn unescape_field(value: &str) -> String {
585    value
586        .replace("%3B", ";")
587        .replace("%7C", "|")
588        .replace("%25", "%")
589}
