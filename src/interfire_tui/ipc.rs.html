15/// Fixed subscription id so reconnect replaces the prior stream.
16pub const AUDIT_SUBSCRIBER_ID: &str = "interfire-tui";
17
18const STATUS_INTERVAL: Duration = Duration::from_secs(1);
19const LIST_INTERVAL: Duration = Duration::from_secs(2);
20const RECONNECT_BACKOFF: Duration = Duration::from_millis(500);
21
22/// Events pushed to the UI from background IPC tasks.
23#[derive(Clone, Debug, Eq, PartialEq)]
24pub enum IpcEvent {
25    Status(DaemonStatus),
26    Down(String),
27    Audit(AuditStreamRecord),
28    SubscriptionReady,
29    Rules(Vec<RuleRow>),
30    Prompts(Vec<PromptRow>),
31    ActionOk(String),
32    ActionError(String),
33}
34
35/// Commands from the UI to the IPC worker.
36#[derive(Clone, Debug, Eq, PartialEq)]
37pub enum IpcCommand {
38    RefreshRules,
39    RefreshPrompts,
40    DeleteRule {
41        id: u64,
42    },
43    AddRule {
44        id: u64,
45        executable: String,
46        verdict: String,
47        port: u16,
48    },
49    AnswerPrompt {
50        id: u64,
51        verdict: String,
52        scope: String,
53    },
54}
55
56/// Spawn non-blocking status, audit, list poll, and command loops.
57pub fn spawn(
58    socket: String,
59    tx: mpsc::UnboundedSender<IpcEvent>,
60    mut commands: mpsc::UnboundedReceiver<IpcCommand>,
61) {
62    let status_socket = socket.clone();
63    let status_tx = tx.clone();
64    tokio::spawn(async move {
65        status_loop(status_socket, status_tx).await;
66    });
67    let audit_socket = socket.clone();
68    let audit_tx = tx.clone();
69    tokio::spawn(async move {
70        audit_loop(audit_socket, audit_tx).await;
71    });
72    let lists_socket = socket.clone();
73    let lists_tx = tx.clone();
74    tokio::spawn(async move {
75        lists_poll_loop(lists_socket, lists_tx).await;
76    });
77    tokio::spawn(async move {
78        while let Some(command) = commands.recv().await {
79            handle_command(&socket, &tx, command).await;
80        }
81    });
82}

84async fn status_loop(socket: String, tx: mpsc::UnboundedSender<IpcEvent>) {
85    let mut interval = time::interval(STATUS_INTERVAL);
86    loop {
87        interval.tick().await;
88        match fetch_status(&socket).await {
89            Ok(status) => {
90                if tx.send(IpcEvent::Status(status)).is_err() {
91                    return;
92                }
93            }
94            Err(error) => {
95                if tx.send(IpcEvent::Down(error.to_string())).is_err() {
96                    return;
97                }
98            }
99        }
100    }
101}
102
103async fn lists_poll_loop(socket: String, tx: mpsc::UnboundedSender<IpcEvent>) {
104    let mut interval = time::interval(LIST_INTERVAL);
105    loop {
106        interval.tick().await;
107        if let Ok(rules) = fetch_rules(&socket).await
108            && tx.send(IpcEvent::Rules(rules)).is_err()
109        {
110            return;
111        }
112        if let Ok(prompts) = fetch_prompts(&socket).await
113            && tx.send(IpcEvent::Prompts(prompts)).is_err()
114        {
115            return;
116        }
117    }
118}

120async fn audit_loop(socket: String, tx: mpsc::UnboundedSender<IpcEvent>) {
121    let mut since = 0_u64;
122    loop {
123        if let Ok(next_since) = subscribe_session(&socket, since, &tx).await {
124            since = next_since;
125        }
126        time::sleep(RECONNECT_BACKOFF).await;
127    }
128}

130async fn handle_command(socket: &str, tx: &mpsc::UnboundedSender<IpcEvent>, command: IpcCommand) {
131    match command {
132        IpcCommand::RefreshRules => match fetch_rules(socket).await {
133            Ok(rules) => {
134                let _ = tx.send(IpcEvent::Rules(rules));
135            }
136            Err(error) => {
137                let _ = tx.send(IpcEvent::ActionError(error.to_string()));
138            }
139        },
140        IpcCommand::RefreshPrompts => match fetch_prompts(socket).await {
141            Ok(prompts) => {
142                let _ = tx.send(IpcEvent::Prompts(prompts));
143            }
144            Err(error) => {
145                let _ = tx.send(IpcEvent::ActionError(error.to_string()));
146            }
147        },
148        IpcCommand::DeleteRule { id } => {
149            mutate_then_refresh_rules(
150                socket,
151                tx,
152                &format!("v1 rule-delete {id}\n"),
153                format!("deleted rule {id}"),
154            )
155            .await;
156        }
157        IpcCommand::AddRule {
158            id,
159            executable,
160            verdict,
161            port,
162        } => {
163            let request = format!("v1 rule-add {id} {executable} {verdict} {port}\n");
164            mutate_then_refresh_rules(socket, tx, &request, format!("added rule {id}")).await;
165        }
166        IpcCommand::AnswerPrompt { id, verdict, scope } => {
167            let request = format!("v1 prompt-answer {id} {verdict} {scope}\n");
168            match one_shot(socket, &request).await {
169                Ok(frame) if frame.starts_with("v1 pong") => {
170                    let _ = tx.send(IpcEvent::ActionOk(format!(
171                        "answered prompt {id} {verdict}/{scope}"
172                    )));
173                    if let Ok(prompts) = fetch_prompts(socket).await {
174                        let _ = tx.send(IpcEvent::Prompts(prompts));
175                    }
176                }
177                Ok(frame) => {
178                    let message =
179                        parse_error_message(&frame).unwrap_or_else(|| frame.trim().to_owned());
180                    let _ = tx.send(IpcEvent::ActionError(message));
181                }
182                Err(error) => {
183                    let _ = tx.send(IpcEvent::ActionError(error.to_string()));
184                }
185            }
186        }
187    }
188}

190async fn mutate_then_refresh_rules(
191    socket: &str,
192    tx: &mpsc::UnboundedSender<IpcEvent>,
193    request: &str,
194    ok: String,
195) {
196    match one_shot(socket, request).await {
197        Ok(frame) if frame.starts_with("v1 pong") => {
198            let _ = tx.send(IpcEvent::ActionOk(ok));
199            if let Ok(rules) = fetch_rules(socket).await {
200                let _ = tx.send(IpcEvent::Rules(rules));
201            }
202        }
203        Ok(frame) => {
204            let message = parse_error_message(&frame).unwrap_or_else(|| frame.trim().to_owned());
205            let _ = tx.send(IpcEvent::ActionError(message));
206        }
207        Err(error) => {
208            let _ = tx.send(IpcEvent::ActionError(error.to_string()));
209        }
210    }
211}

213async fn fetch_status(socket: &str) -> io::Result<DaemonStatus> {
214    let frame = one_shot(socket, "v1 status\n").await?;
215    DaemonStatus::parse(&frame)
216        .map_err(|_| io::Error::new(io::ErrorKind::InvalidData, "malformed_status"))
217}

219async fn fetch_rules(socket: &str) -> io::Result<Vec<RuleRow>> {
220    let frame = one_shot(socket, "v1 rule-list\n").await?;
221    RuleRow::parse_frame(&frame)

231async fn one_shot(socket: &str, request: &str) -> io::Result<String> {
232    let stream = UnixStream::connect(socket).await?;
233    let (reader, mut writer) = stream.into_split();
234    writer.write_all(request.as_bytes()).await?;
235    let mut line = String::new();
236    BufReader::new(reader).read_line(&mut line).await?;
237    if line.len() > MAX_FRAME_BYTES {
238        return Err(io::Error::new(
239            io::ErrorKind::InvalidData,
240            "frame_too_large",
241        ));
242    }
243    Ok(line)
244}

246async fn subscribe_session(
247    socket: &str,
248    since: u64,
249    tx: &mpsc::UnboundedSender<IpcEvent>,
250) -> io::Result<u64> {
251    let stream = UnixStream::connect(socket).await?;
252    let (reader, mut writer) = stream.into_split();
253    let request = format!("v1 audit-subscribe {AUDIT_SUBSCRIBER_ID} since={since}\n");
254    writer.write_all(request.as_bytes()).await?;
255    let mut lines = BufReader::new(reader).lines();
256    let mut last_since = since;
257    while let Some(line) = lines.next_line().await? {
258        if line.len() > MAX_FRAME_BYTES {
259            return Err(io::Error::new(
260                io::ErrorKind::InvalidData,
261                "frame_too_large",
262            ));
263        }
264        if line.starts_with("v1 subscribed ") {
265            if tx.send(IpcEvent::SubscriptionReady).is_err() {
266                return Ok(last_since);
267            }
268            continue;
269        }
270        if line == "v1 audit-replaced" {
271            continue;
272        }
273        match AuditStreamRecord::parse(&line) {
274            Ok(record) => {
275                last_since = record.sequence;
276                if tx.send(IpcEvent::Audit(record)).is_err() {
277                    return Ok(last_since);
278                }
279            }
280            Err(_) => {
281                return Err(io::Error::new(
282                    io::ErrorKind::InvalidData,
283                    "malformed_audit",
284                ));
285            }
286        }
287    }
288    Ok(last_since)
289}
