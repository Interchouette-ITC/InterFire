109#[derive(Clone, Debug, Eq, PartialEq)]
110pub struct AddRuleForm {
111    pub id: String,
112    pub executable: String,
113    pub verdict: String,
114    pub port: String,
115    pub focus: AddField,
116}
117
118impl AddRuleForm {
119    fn new() -> Self {
120        Self {
121            id: String::new(),
122            executable: String::new(),
123            verdict: "deny".into(),
124            port: "443".into(),
125            focus: AddField::Id,
126        }
127    }
128
129    const fn focused_mut(&mut self) -> &mut String {
130        match self.focus {
131            AddField::Id => &mut self.id,
132            AddField::Executable => &mut self.executable,
133            AddField::Verdict => &mut self.verdict,
134            AddField::Port => &mut self.port,
135        }
136    }
137
138    fn to_command(&self) -> Result<IpcCommand, String> {
139        let id = self
140            .id
141            .parse::<u64>()
142            .map_err(|_| "id must be an integer".to_owned())?;
143        if self.executable.is_empty() || !self.executable.starts_with('/') {
144            return Err("executable must be an absolute path".into());
145        }
146        let verdict = self.verdict.to_ascii_lowercase();
147        if !matches!(verdict.as_str(), "allow" | "deny" | "prompt") {
148            return Err("verdict must be allow|deny|prompt".into());
149        }
150        let port = self
151            .port
152            .parse::<u16>()
153            .map_err(|_| "port must be 0..65535".to_owned())?;
154        Ok(IpcCommand::AddRule {
155            id,
156            executable: self.executable.clone(),
157            verdict,
158            port,
159        })
160    }
161}

301    pub fn apply(&mut self, event: IpcEvent) {
302        match event {
303            IpcEvent::Status(status) => {
304                self.link = Link::Up(status);
305            }
306            IpcEvent::Down(detail) => {
307                self.link = Link::Down { detail };
308                self.subscribed = false;
309            }
310            IpcEvent::Audit(record) => {
311                if self.audit.len() == MAX_AUDIT_LINES {
312                    self.audit.pop_front();
313                }
314                self.audit
315                    .push_back(format!("{}|{}", record.sequence, record.message));
316                self.clamp_selection();
317            }
318            IpcEvent::SubscriptionReady => {
319                self.subscribed = true;
320            }
321            IpcEvent::Rules(rules) => {
322                self.rules = rules;
323                self.clamp_selection();
324            }
325            IpcEvent::Prompts(prompts) => {
326                self.prompts = prompts;
327                self.clamp_selection();
328            }
329            IpcEvent::ActionOk(message) => {
330                self.status_message = Some(message);
331            }
332            IpcEvent::ActionError(message) => {
333                self.overlay = Overlay::Notice(format!("error: {message}"));
334                self.status_message = Some(format!("error: {message}"));
335            }
336        }
337    }

353    /// Handle a key at root chrome or overlay.
354    pub fn handle_key(&mut self, code: KeyCode) -> KeyAction {
355        match self.overlay {
356            Overlay::None => self.handle_root_key(code),
357            Overlay::Notice(_) => {
358                if matches!(code, KeyCode::Esc | KeyCode::Enter | KeyCode::Char('q')) {
359                    self.overlay = Overlay::None;
360                }
361                KeyAction::None
362            }
363            Overlay::AddRule(_) => self.handle_add_overlay_key(code),
364            Overlay::AnswerPrompt(_) => self.handle_answer_overlay_key(code),
365        }
366    }
367
368    fn handle_add_overlay_key(&mut self, code: KeyCode) -> KeyAction {
369        match code {
370            KeyCode::Esc => {
371                self.overlay = Overlay::None;
372                return KeyAction::None;
373            }
374            KeyCode::Enter => {
375                let command = match &self.overlay {
376                    Overlay::AddRule(form) => form.to_command(),
377                    Overlay::None | Overlay::Notice(_) | Overlay::AnswerPrompt(_) => {
378                        return KeyAction::None;
379                    }
380                };
381                return match command {
382                    Ok(command) => {
383                        self.overlay = Overlay::None;
384                        KeyAction::Command(command)
385                    }
386                    Err(message) => {
387                        self.status_message = Some(message);
388                        KeyAction::None
389                    }
390                };
391            }
392            _ => {}
393        }
394        let Overlay::AddRule(form) = &mut self.overlay else {
395            return KeyAction::None;
396        };
397        match code {
398            KeyCode::Tab => {
399                form.focus = form.focus.next();
400            }
401            KeyCode::BackTab => {
402                form.focus = form.focus.prev();
403            }
404            KeyCode::Backspace => {
405                form.focused_mut().pop();
406            }
407            KeyCode::Char(ch) if !ch.is_control() => {
408                form.focused_mut().push(ch);
409            }
410            _ => {}
411        }
412        KeyAction::None
413    }
414
415    fn handle_answer_overlay_key(&mut self, code: KeyCode) -> KeyAction {
416        match code {
417            KeyCode::Esc => {
418                self.overlay = Overlay::None;
419                return KeyAction::None;
420            }
421            KeyCode::Enter => {
422                let command = match &self.overlay {
423                    Overlay::AnswerPrompt(form) => {
424                        if !form.prompt.can_answer() {
425                            self.overlay =
426                                Overlay::Notice("prompt expired or stale; actions disabled".into());
427                            return KeyAction::None;
428                        }
429                        form.to_command()
430                    }
431                    Overlay::None | Overlay::Notice(_) | Overlay::AddRule(_) => {
432                        return KeyAction::None;
433                    }
434                };
435                self.overlay = Overlay::None;
436                return KeyAction::Command(command);
437            }
438            _ => {}
439        }
440        let Overlay::AnswerPrompt(form) = &mut self.overlay else {
441            return KeyAction::None;
442        };
443        match code {
444            KeyCode::Char('a' | 'A') => form.verdict = AnswerVerdict::Allow,
445            KeyCode::Char('d' | 'D') => form.verdict = AnswerVerdict::Deny,
446            KeyCode::Left | KeyCode::Right => form.verdict = form.verdict.toggle(),
447            KeyCode::Tab => form.scope = form.scope.next(),
448            KeyCode::BackTab => form.scope = form.scope.prev(),
449            _ => {}
450        }
451        KeyAction::None
452    }
453
454    fn handle_root_key(&mut self, code: KeyCode) -> KeyAction {
455        match code {
456            KeyCode::Char('q' | 'Q') => KeyAction::Quit,
457            KeyCode::Char('?') => {
458                self.overlay = Overlay::Notice("Press Esc to dismiss this overlay.".into());
459                KeyAction::None
460            }
461            KeyCode::Left | KeyCode::BackTab => {
462                self.prev_tab();
463                KeyAction::None
464            }
465            KeyCode::Right | KeyCode::Tab => {
466                self.next_tab();
467                KeyAction::None
468            }
469            KeyCode::Char('1') => {
470                self.set_tab(Tab::Status);
471                KeyAction::None
472            }
473            KeyCode::Char('2') => {
474                self.set_tab(Tab::Rules);
475                KeyAction::None
476            }
477            KeyCode::Char('3') => {
478                self.set_tab(Tab::Prompts);
479                KeyAction::None
480            }
481            KeyCode::Char('4') => {
482                self.set_tab(Tab::Log);
483                KeyAction::None
484            }
485            KeyCode::Char('5') => {
486                self.set_tab(Tab::Help);
487                KeyAction::None
488            }
489            KeyCode::Char('h') if self.tab.has_split() => {
490                self.pane = Pane::List;
491                KeyAction::None
492            }
493            KeyCode::Char('l') if self.tab.has_split() => {
494                self.pane = Pane::Detail;
495                KeyAction::None
496            }
497            KeyCode::Char('j') | KeyCode::Down if self.pane == Pane::List => {
498                self.move_list(1);
499                KeyAction::None
500            }
501            KeyCode::Char('k') | KeyCode::Up if self.pane == Pane::List => {
502                self.move_list(-1);
503                KeyAction::None
504            }
505            KeyCode::Char('a') if self.tab == Tab::Rules => {
506                self.overlay = Overlay::AddRule(AddRuleForm::new());
507                KeyAction::None
508            }
509            KeyCode::Char('d') if self.tab == Tab::Rules => self.delete_selected_rule(),
510            KeyCode::Char('r') if self.tab == Tab::Rules => {
511                KeyAction::Command(IpcCommand::RefreshRules)
512            }
513            KeyCode::Char('a') | KeyCode::Enter if self.tab == Tab::Prompts => {
514                self.open_answer_overlay()
515            }
516            KeyCode::Char('r') if self.tab == Tab::Prompts => {
517                KeyAction::Command(IpcCommand::RefreshPrompts)
518            }
519            _ => KeyAction::None,
520        }
521    }
522
523    fn open_answer_overlay(&mut self) -> KeyAction {
524        match self.prompts.get(self.list_selected) {
525            Some(prompt) if prompt.can_answer() => {
526                self.overlay = Overlay::AnswerPrompt(AnswerPromptForm::new(prompt.clone()));
527                KeyAction::None
528            }
529            Some(_) => {
530                self.overlay = Overlay::Notice("prompt expired or stale; answer disabled".into());
531                KeyAction::None
532            }
533            None => KeyAction::None,
534        }
535    }
536
537    fn delete_selected_rule(&self) -> KeyAction {
538        self.rules
539            .get(self.list_selected)
540            .map_or(KeyAction::None, |rule| {
541                KeyAction::Command(IpcCommand::DeleteRule { id: rule.id })
542            })
543    }
544
545    fn set_tab(&mut self, tab: Tab) {
546        self.tab = tab;
547        self.pane = Pane::List;
548        self.list_selected = 0;
549        self.clamp_selection();
550    }
551
552    fn next_tab(&mut self) {
553        self.set_tab(Tab::from_index(self.tab.index().saturating_add(1)));
554    }
555
556    fn prev_tab(&mut self) {
557        let index = if self.tab.index() == 0 {
558            Tab::ALL.len() - 1
559        } else {
560            self.tab.index() - 1
561        };
562        self.set_tab(Tab::from_index(index));
563    }
564
565    fn move_list(&mut self, delta: isize) {
566        let len = self.list_len();
567        if len == 0 {
568            self.list_selected = 0;
569            return;
570        }
571        let current = isize::try_from(self.list_selected).unwrap_or(0);
572        let next = (current + delta).rem_euclid(isize::try_from(len).unwrap_or(1));
573        self.list_selected = usize::try_from(next).unwrap_or(0);
574    }
575
576    fn clamp_selection(&mut self) {
577        let len = self.list_len();
578        if len == 0 {
579            self.list_selected = 0;
580        } else if self.list_selected >= len {
581            self.list_selected = len - 1;
582        }
583    }

637        self.prompts.get(self.list_selected).map_or_else(
638            || vec!["no pending prompts".into(), "r refresh".into()],

641                    format!("id: {}", prompt.id),
642                    format!("path: {}", prompt.executable),
643                    format!(
644                        "dest: {}:{} ({})",

647                    format!("remaining: {}s", prompt.remaining_secs),
648                ];
649                lines.push(String::new());
650                if prompt.can_answer() {
651                    lines.push("a/Enter answer · r refresh".into());
652                } else {
653                    lines.push("expired/stale · answer disabled".into());
654                }
655                lines
656            },
657        )
