1//! Ratatui drawing for tab chrome, panes, footer, and overlay.
2#![forbid(unsafe_code)]
3
4use interfire_proto::IPC_VERSION;
5use ratatui::Frame;
6use ratatui::layout::{Constraint, Direction, Layout, Rect};
7use ratatui::style::{Modifier, Style};
8use ratatui::text::{Line, Span};
9use ratatui::widgets::{Block, Borders, Clear, List, ListItem, ListState, Paragraph, Wrap};
10
11use crate::app::{App, Overlay, Pane, Tab};
12
13pub fn draw(frame: &mut Frame<'_>, app: &App) {
14    let chunks = Layout::vertical([
15        Constraint::Length(3),
16        Constraint::Min(3),
17        Constraint::Length(1),
18    ])
19    .split(frame.area());
20    frame.render_widget(tabs_bar(app), chunks[0]);
21    draw_body(frame, app, chunks[1]);
22    frame.render_widget(Paragraph::new(crate::app::footer_hints(app)), chunks[2]);
23    if let Overlay::Notice(message) = app.overlay {
24        draw_overlay(frame, message);
25    }
26}

28fn tabs_bar(app: &App) -> Paragraph<'static> {
29    let mut spans = vec![
30        Span::styled("InterFire", Style::default().add_modifier(Modifier::BOLD)),
31        Span::raw(format!(" v{IPC_VERSION}  ")),
32    ];
33    for tab in Tab::ALL {
34        let label = format!(" {} ", tab.label());
35        if tab == app.tab {
36            spans.push(Span::styled(
37                label,
38                Style::default().add_modifier(Modifier::REVERSED | Modifier::BOLD),
39            ));
40        } else {
41            spans.push(Span::raw(label));
42        }
43    }
44    Paragraph::new(Line::from(spans)).block(
45        Block::default()
46            .borders(Borders::ALL)
47            .title(format!("tabs · {}", app.chrome_title())),
48    )
49}

51fn draw_body(frame: &mut Frame<'_>, app: &App, area: Rect) {
52    if app.tab.has_split() {
53        let panes = Layout::default()
54            .direction(Direction::Horizontal)
55            .constraints([Constraint::Percentage(45), Constraint::Percentage(55)])
56            .split(area);
57        let mut state = list_state(app);
58        frame.render_stateful_widget(list_widget(app), panes[0], &mut state);
59        frame.render_widget(detail_widget(app), panes[1]);
60    } else {
61        frame.render_widget(detail_widget(app), area);
62    }
63}
64
65fn list_state(app: &App) -> ListState {
66    let mut state = ListState::default();
67    if app.list_len() > 0 {
68        state.select(Some(app.list_selected));
69    }
70    state
71}

73fn list_widget(app: &App) -> List<'static> {
74    let items: Vec<ListItem<'static>> = app.list_items().into_iter().map(ListItem::new).collect();
75    let title = if app.pane == Pane::List {
76        format!("{} list [focus]", app.tab.label())
77    } else {
78        format!("{} list", app.tab.label())
79    };
80    List::new(items)
81        .block(Block::default().borders(Borders::ALL).title(title))
82        .highlight_style(Style::default().add_modifier(Modifier::REVERSED))
83}
84
85fn detail_widget(app: &App) -> Paragraph<'_> {
86    let title = if app.tab.has_split() && app.pane == Pane::Detail {
87        format!("{} detail [focus]", app.tab.label())
88    } else {
89        format!("{} detail", app.tab.label())
90    };
91    let lines: Vec<Line<'_>> = app.detail_lines().into_iter().map(Line::from).collect();
92    Paragraph::new(lines)
93        .wrap(Wrap { trim: false })
94        .block(Block::default().borders(Borders::ALL).title(title))
95}

97fn draw_overlay(frame: &mut Frame<'_>, message: &str) {
98    let area = centered_rect(60, 30, frame.area());
99    frame.render_widget(Clear, area);
100    frame.render_widget(
101        Paragraph::new(vec![
102            Line::from(Span::styled(
103                "Overlay",
104                Style::default().add_modifier(Modifier::BOLD),
105            )),
106            Line::from(""),
107            Line::from(message),
108            Line::from(""),
109            Line::from("Esc dismisses · does not quit"),
110        ])
111        .block(Block::default().borders(Borders::ALL).title("notice")),
112        area,
113    );
114}

116fn centered_rect(percent_x: u16, percent_y: u16, area: Rect) -> Rect {
117    let vertical = Layout::vertical([
118        Constraint::Percentage((100 - percent_y) / 2),
119        Constraint::Percentage(percent_y),
120        Constraint::Percentage((100 - percent_y) / 2),
121    ])
122    .split(area);
123    Layout::horizontal([
124        Constraint::Percentage((100 - percent_x) / 2),
125        Constraint::Percentage(percent_x),
126        Constraint::Percentage((100 - percent_x) / 2),
127    ])
128    .split(vertical[1])[1]
129}
