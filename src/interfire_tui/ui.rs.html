15pub fn draw(frame: &mut Frame<'_>, app: &App) {
16    let chunks = Layout::vertical([
17        Constraint::Length(3),
18        Constraint::Min(3),
19        Constraint::Length(2),
20    ])
21    .split(frame.area());
22    frame.render_widget(tabs_bar(app), chunks[0]);
23    draw_body(frame, app, chunks[1]);
24    frame.render_widget(
25        Paragraph::new(vec![
26            Line::from(stats_footer_line(app)),
27            Line::from(crate::app::footer_hints(app)).style(palette::muted()),
28        ]),
29        chunks[2],
30    );
31    match &app.overlay {
32        Overlay::None => {}
33        Overlay::Notice(message) => draw_notice(frame, message),
34        Overlay::AddRule(form) => draw_add_rule(frame, form),
35        Overlay::AnswerPrompt(form) => draw_answer_prompt(frame, form),
36        Overlay::Traffic(form) => draw_traffic(frame, *form),
37    }
38}

40fn stats_footer_line(app: &App) -> String {
41    let filter = app.events_filter.label();
42    app.stats_summary.as_ref().map_or_else(
43        || {
44            format!(
45                "Connections -  ·  Denied -  ·  Uptime -  ·  Rules -  ·  Version -  ·  filter {filter}"
46            )
47        },
48        |summary| {
49            format!(
50                "Connections {}  ·  Denied {}  ·  Uptime {}s  ·  Rules {}  ·  Version {}+{}  ·  filter {filter}",
51                summary.connections,
52                summary.denied,
53                summary.uptime_secs,
54                summary.rules,
55                summary.version,
56                summary.git,
57            )
58        },
59    )
60}

62fn tabs_bar(app: &App) -> Paragraph<'static> {
63    let mut spans = vec![
64        Span::styled("InterFire", palette::title()),
65        Span::styled(format!(" v{IPC_VERSION}  "), palette::muted()),
66    ];
67    for tab in Tab::ALL {
68        let label = format!(" {} ", tab.label());
69        if tab == app.tab {
70            spans.push(Span::styled(label, palette::selected()));
71        } else {
72            spans.push(Span::styled(label, palette::body()));
73        }
74    }
75    Paragraph::new(Line::from(spans)).block(
76        Block::default()
77            .borders(Borders::ALL)
78            .border_style(palette::border())
79            .style(palette::chrome())
80            .title(Span::styled(
81                format!("tabs · {}", app.chrome_title()),
82                palette::muted(),
83            )),
84    )
85}

87fn draw_body(frame: &mut Frame<'_>, app: &App, area: Rect) {
88    if app.tab.has_split() {
89        let panes = Layout::default()
90            .direction(Direction::Horizontal)
91            .constraints([Constraint::Percentage(45), Constraint::Percentage(55)])
92            .split(area);
93        let viewport_rows = usize::from(panes[0].height.saturating_sub(2));
94        let visible = app.visible_list(viewport_rows);
95        let mut state = ListState::default();
96        if !visible.items.is_empty() {
97            state.select(Some(visible.relative_selected));
98        }
99        frame.render_stateful_widget(list_widget(app, &visible), panes[0], &mut state);
100        frame.render_widget(detail_widget(app), panes[1]);
101    } else {
102        frame.render_widget(detail_widget(app), area);
103    }
104}

106fn list_widget(app: &App, visible: &crate::app::VisibleList) -> List<'static> {
107    let items: Vec<ListItem<'static>> = visible
108        .items
109        .iter()
110        .cloned()
111        .map(|line| ListItem::new(line).style(palette::body()))
112        .collect();
113    let focus = if app.pane == Pane::List {
114        " [focus]"
115    } else {
116        ""
117    };
118    let title = if app.tab == Tab::Events && visible.total > 0 {
119        let shown = visible.start.saturating_add(1);
120        let end = visible.start.saturating_add(visible.items.len());
121        format!(
122            "Events list{focus}  {shown}-{end}/{total}",
123            total = visible.total
124        )
125    } else {
126        format!("{} list{focus}", app.tab.label())
127    };
128    List::new(items)
129        .block(
130            Block::default()
131                .borders(Borders::ALL)
132                .border_style(palette::border())
133                .style(palette::panel())
134                .title(Span::styled(title, palette::muted())),
135        )
136        .highlight_style(palette::selected())
137}

139fn detail_widget(app: &App) -> Paragraph<'_> {
140    let title = if app.tab.has_split() && app.pane == Pane::Detail {
141        format!("{} detail [focus]", app.tab.label())
142    } else {
143        format!("{} detail", app.tab.label())
144    };
145    let lines: Vec<Line<'_>> = app
146        .detail_lines()
147        .into_iter()
148        .map(|line| Line::from(Span::styled(line, palette::body())))
149        .collect();
150    Paragraph::new(lines)
151        .wrap(Wrap { trim: false })
152        .style(palette::panel())
153        .block(
154            Block::default()
155                .borders(Borders::ALL)
156                .border_style(palette::border())
157                .title(Span::styled(title, palette::muted())),
158        )
159}

161fn draw_notice(frame: &mut Frame<'_>, message: &str) {
162    let area = centered_rect(60, 30, frame.area());
163    frame.render_widget(Clear, area);
164    frame.render_widget(
165        Paragraph::new(vec![
166            Line::from(Span::styled("Overlay", palette::title())),
167            Line::from(""),
168            Line::from(Span::styled(message.to_owned(), palette::body())),
169            Line::from(""),
170            Line::from(Span::styled(
171                "Esc dismisses · does not quit",
172                palette::muted(),
173            )),
174        ])
175        .style(palette::overlay_panel())
176        .block(
177            Block::default()
178                .borders(Borders::ALL)
179                .border_style(palette::border())
180                .title(Span::styled("notice", palette::accent_label())),
181        ),
182        area,
183    );
184}

186fn draw_add_rule(frame: &mut Frame<'_>, form: &AddRuleForm) {
187    let area = centered_rect(70, 55, frame.area());
188    frame.render_widget(Clear, area);
189    let lines = vec![
190        field_line(form, AddField::Id, &form.id),
191        field_line(form, AddField::Executable, &form.executable),
192        field_line(form, AddField::Verdict, &form.verdict),
193        field_line(form, AddField::Port, &form.port),
194        Line::from(""),
195        Line::from(Span::styled(
196            "Tab next field · Enter submit · Esc cancel",
197            palette::muted(),
198        )),
199    ];
200    frame.render_widget(
201        Paragraph::new(lines).style(palette::overlay_panel()).block(
202            Block::default()
203                .borders(Borders::ALL)
204                .border_style(palette::border())
205                .title(Span::styled("add rule", palette::accent_label())),
206        ),
207        area,
208    );
209}

211fn draw_answer_prompt(frame: &mut Frame<'_>, form: &AnswerPromptForm) {
212    let area = centered_rect(70, 60, frame.area());
213    frame.render_widget(Clear, area);
214    let prompt = &form.prompt;
215    let allow_style = if form.verdict == AnswerVerdict::Allow {
216        palette::allow()
217    } else {
218        palette::body()
219    };
220    let deny_style = if form.verdict == AnswerVerdict::Deny {
221        palette::deny()
222    } else {
223        palette::body()
224    };
225    let remaining_style = if prompt.remaining_secs <= 5 {
226        palette::err()
227    } else if prompt.remaining_secs <= 15 {
228        palette::warn()
229    } else {
230        palette::ok()
231    };
232    let lines = vec![
233        Line::from(Span::styled(
234            format!("prompt #{}", prompt.id),
235            palette::title(),
236        )),
237        Line::from(Span::styled(
238            format!("path: {}", prompt.executable),
239            palette::body(),
240        )),
241        Line::from(Span::styled(
242            format!(
243                "dest: {}:{} ({})",
244                prompt.destination, prompt.port, prompt.protocol
245            ),
246            palette::body(),
247        )),
248        Line::from(Span::styled(
249            format!("remaining: {}s", prompt.remaining_secs),
250            remaining_style,
251        )),
252        Line::from(""),
253        Line::from(vec![
254            Span::styled("verdict: ", palette::muted()),
255            Span::styled(" allow ", allow_style),
256            Span::styled(" deny ", deny_style),
257        ]),
258        Line::from(Span::styled(
259            format!(
260                "scope:   {}  (Tab cycles once|session|permanent)",
261                form.scope.as_str()
262            ),
263            palette::body(),
264        )),
265        Line::from(""),
266        Line::from(Span::styled(
267            "a/d or Left/Right verdict · Enter submit · Esc cancel",
268            palette::muted(),
269        )),
270    ];
271    frame.render_widget(
272        Paragraph::new(lines).style(palette::overlay_panel()).block(
273            Block::default()
274                .borders(Borders::ALL)
275                .border_style(palette::border())
276                .title(Span::styled("answer prompt", palette::accent_label())),
277        ),
278        area,
279    );
280}

282fn draw_traffic(frame: &mut Frame<'_>, form: TrafficForm) {
283    let area = centered_rect(70, 50, frame.area());
284    frame.render_widget(Clear, area);
285    let lines = vec![

292        )),

294            format!(

296                form.direction.as_str()
297            ),
298            palette::body(),
299        )),
300        Line::from(Span::styled(
301            format!(
302                "action:    {}  (b block · u unblock · a toggle)",
303                form.action.as_str()
304            ),
305            palette::body(),
306        )),
307        Line::from(""),
308        Line::from(Span::styled(
309            "Machine scope: use pkexec CLI (TUI does not elevate)",

316            Block::default()
317                .borders(Borders::ALL)
318                .border_style(palette::border())

320        ),
321        area,
322    );
323}

325fn field_line(form: &AddRuleForm, field: AddField, value: &str) -> Line<'static> {
326    let marker = if form.focus == field { ">" } else { " " };
327    let style = if form.focus == field {
328        palette::focus()
329    } else {
330        palette::body()
331    };
332    Line::from(Span::styled(
333        format!("{marker} {}: {value}", field.label()),
334        style,
335    ))
336}

338fn centered_rect(percent_x: u16, percent_y: u16, area: Rect) -> Rect {
339    let vertical = Layout::vertical([
340        Constraint::Percentage((100 - percent_y) / 2),
341        Constraint::Percentage(percent_y),
342        Constraint::Percentage((100 - percent_y) / 2),
343    ])
344    .split(area);
345    Layout::horizontal([
346        Constraint::Percentage((100 - percent_x) / 2),
347        Constraint::Percentage(percent_x),
348        Constraint::Percentage((100 - percent_x) / 2),
349    ])
350    .split(vertical[1])[1]
351}
