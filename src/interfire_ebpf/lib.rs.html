1//! TCP-connect observation contract and userspace loader.
2//!
3//! The eBPF program emits bounded [`TcpConnectEvent`] records. Attribution and
4//! verdicts happen in userspace (`/proc`, rules, NFQUEUE).
5#![cfg_attr(not(feature = "loader"), no_std)]
6#![cfg_attr(feature = "loader", forbid(unsafe_code))]
7
8#[cfg(feature = "loader")]
9mod loader;
10
11#[cfg(feature = "loader")]
12pub use loader::{EVENT_MAP, LoadError, Observer, ObserverStatus, PROGRAM_NAME};
13
14/// Ring buffer capacity in bytes (power-of-two page multiple; 256 KiB).
15pub const RINGBUF_BYTE_SIZE: u32 = 256 * 1024;
16
17/// Bounded wire payload emitted by the TCP-connect program.
18///
19/// `pid` is the thread-group id (TGID) suitable for `/proc/<pid>`.
20/// `process_start_ticks` is always zero from the kernel program; userspace fills
21/// start ticks from `/proc` immediately after dequeue.
22#[derive(Clone, Copy, Debug, Eq, PartialEq)]
23#[repr(C)]
24pub struct TcpConnectEvent {
25    pub pid: u32,
26    pub process_start_ticks: u64,
27    pub destination_ipv4: u32,
28    pub destination_port: u16,
29    pub reserved: u16,
30}
31
32const _: () = assert!(core::mem::size_of::<TcpConnectEvent>() == 24);
33const _: () = assert!(core::mem::align_of::<TcpConnectEvent>() <= 8);
34
35impl TcpConnectEvent {
36    /// Destination as an IPv4 address in host byte order octets.
37    #[must_use]
38    pub const fn destination_octets(self) -> [u8; 4] {
39        self.destination_ipv4.to_ne_bytes()
40    }
41}
42
43#[cfg(test)]
44mod tests {
45    use super::*;
46
47    #[test]
48    fn event_layout_is_stable() {
49        assert_eq!(core::mem::size_of::<TcpConnectEvent>(), 24);
50        assert!(core::mem::align_of::<TcpConnectEvent>() <= 8);
51    }
52
53    #[test]
54    fn destination_octets_match_host_order_bytes() {
55        let event = TcpConnectEvent {
56            pid: 1,
57            process_start_ticks: 0,
58            destination_ipv4: u32::from_ne_bytes([127, 0, 0, 1]),
59            destination_port: 443,
60            reserved: 0,
61        };
62        assert_eq!(event.destination_octets(), [127, 0, 0, 1]);
63    }
64}
