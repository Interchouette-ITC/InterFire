1//! Load, attach, and detach the TCP-connect observation program.
2
3use std::io;
4use std::path::Path;
5
6use aya::maps::{MapData, MapError, RingBuf};
7use aya::programs::ProgramError;
8use aya::{Ebpf, EbpfError, EbpfLoader};
9
10/// eBPF program section / function name attached to `tcp_v4_connect`.
11pub const PROGRAM_NAME: &str = "interfire_tcp_connect";
12
13/// Ring buffer map name declared by the eBPF program.
14pub const EVENT_MAP: &str = "EVENTS";
15
16/// Why observation could not start.
17#[derive(Debug, thiserror::Error)]
18pub enum LoadError {
19    /// Embedded or on-disk bytecode was missing or unreadable.
20    #[error("eBPF bytecode unavailable: {0}")]
21    Bytecode(#[source] io::Error),
22    /// `aya` rejected the object (BTF, verifier, or format).
23    #[error("eBPF load failed: {0}")]
24    Ebpf(#[from] EbpfError),
25    /// Map conversion failed.
26    #[error("eBPF map error: {0}")]
27    Map(#[from] MapError),
28    /// Required map or program name was absent from the object.
29    #[error("eBPF object missing {0}")]
30    MissingSymbol(&'static str),
31    /// Attach failed (capabilities, kprobe symbol, or permissions).
32    #[error("eBPF attach failed: {0}")]
33    Attach(#[from] ProgramError),
34}
35
36/// High-level observation readiness for status / CLI.
37#[derive(Clone, Copy, Debug, Eq, PartialEq)]
38pub enum ObserverStatus {
39    /// Kprobe attached; ring buffer readable.
40    Attached,
41    /// Load or attach failed; daemon must not claim enforcement.
42    Degraded,
43}
44
45/// Loaded TCP-connect observer. Dropping detaches with the `Ebpf` object.
46pub struct Observer {
47    pub(crate) bpf: Ebpf,
48}
49
50impl Observer {
51    /// Load bytecode from memory and attach to `tcp_v4_connect`.
52    ///
53    /// `bytecode` must be sufficiently aligned for ELF parsing (use
54    /// [`aya::include_bytes_aligned`] or [`Self::load_path_and_attach`]).
55    ///
56    /// # Errors
57    ///
58    /// Returns [`LoadError`] when the object is invalid, symbols are missing, or
59    /// attach fails (missing caps/BTF/kprobe).
60    pub fn load_and_attach(bytecode: &[u8]) -> Result<Self, LoadError> {
61        let bpf = EbpfLoader::new().load(bytecode)?;
62        Self::attach_loaded(bpf)
63    }
64
65    /// Load bytecode from a filesystem path and attach.
66    ///
67    /// # Errors
68    ///
69    /// Returns [`LoadError`] when the file cannot be read or attach fails.
70    pub fn load_path_and_attach(path: impl AsRef<Path>) -> Result<Self, LoadError> {
71        let bpf = EbpfLoader::new().load_file(path.as_ref())?;
72        Self::attach_loaded(bpf)
73    }
74
75    /// Load the release object embedded beside this crate, then attach.
76    ///
77    /// # Errors
78    ///
79    /// Same as [`Self::load_and_attach`].
80    pub fn load_embedded_and_attach() -> Result<Self, LoadError> {
81        Self::load_and_attach(aya::include_bytes_aligned!(concat!(
82            env!("CARGO_MANIFEST_DIR"),
83            "/bpf/interfire-ebpf-programs"
84        )))
85    }
86
87    fn attach_loaded(bpf: Ebpf) -> Result<Self, LoadError> {
88        #[cfg(not(test))]
89        {
90            crate::loader_attach::attach_loaded(bpf)
91        }
92        #[cfg(test)]
93        {
94            attach_loaded_under_test(bpf)
95        }
96    }
97
98    #[must_use]
99    pub const fn status(&self) -> ObserverStatus {
100        ObserverStatus::Attached
101    }
102
103    /// Borrow the ring buffer map for event polling.
104    ///
105    /// # Errors
106    ///
107    /// Returns [`LoadError::MissingSymbol`] when the map is absent, or
108    /// [`LoadError::Map`] when the map type is wrong.
109    pub fn ring_buf(&mut self) -> Result<RingBuf<&mut MapData>, LoadError> {
110        let map = self
111            .bpf
112            .map_mut(EVENT_MAP)
113            .ok_or(LoadError::MissingSymbol(EVENT_MAP))?;
114        Ok(RingBuf::try_from(map)?)
115    }
116}
117
118#[cfg(test)]
119fn attach_loaded_under_test(bpf: Ebpf) -> Result<Observer, LoadError> {
120    let _ = bpf
121        .map(EVENT_MAP)
122        .ok_or(LoadError::MissingSymbol(EVENT_MAP))?;
123    drop(bpf);
124    Err(LoadError::Bytecode(io::Error::other(
125        "attach skipped under unit tests",
126    )))
127}

155    #[test]
156    fn load_error_display_covers_variants() {
157        let bytecode = LoadError::Bytecode(io::Error::other("missing"));
158        assert!(bytecode.to_string().contains("unavailable"));
159        assert!(bytecode.source().is_some());
160
161        let missing = LoadError::MissingSymbol("EVENTS");
162        assert!(missing.to_string().contains("EVENTS"));
163        assert!(missing.source().is_none());
164    }

166    #[test]
167    fn load_path_missing_file_is_ebpf_error() {
168        let result = Observer::load_path_and_attach("/tmp/interfire-no-such-ebpf-object");
169        assert!(matches!(result, Err(LoadError::Ebpf(_))));
170    }
171
172    #[test]
173    fn load_path_garbage_is_ebpf_error() {
174        let path =
175            std::env::temp_dir().join(format!("interfire-garbage-ebpf-{}", std::process::id()));
176        std::fs::write(&path, b"not-an-elf").unwrap();
177        let result = Observer::load_path_and_attach(&path);
178        let _ = std::fs::remove_file(&path);
179        assert!(matches!(result, Err(LoadError::Ebpf(_))));
180    }

182    #[test]

184        let result = Observer::load_embedded_and_attach();
185        if let Err(error) = result {
186            let message = error.to_string();
187            assert!(
188                message.contains("eBPF") || message.contains("attach") || message.contains("map"),
189                "unexpected error text: {message}"

193    }

222    #[test]
223    fn embedded_object_declares_program_and_event_map() {
224        let file = object::read::File::parse(embedded_bytecode()).expect("elf");
225        let symbols = file
