1//! Load, attach, and detach the TCP-connect observation program.
2
3use std::io;
4use std::path::Path;
5
6use aya::maps::{MapData, MapError, RingBuf};
7use aya::programs::{KProbe, ProgramError};
8use aya::{Ebpf, EbpfError, EbpfLoader};
9
10/// eBPF program section / function name attached to `tcp_v4_connect`.
11pub const PROGRAM_NAME: &str = "interfire_tcp_connect";
12
13/// Ring buffer map name declared by the eBPF program.
14pub const EVENT_MAP: &str = "EVENTS";
15
16/// Why observation could not start.
17#[derive(Debug, thiserror::Error)]
18pub enum LoadError {
19    /// Embedded or on-disk bytecode was missing or unreadable.
20    #[error("eBPF bytecode unavailable: {0}")]
21    Bytecode(#[source] io::Error),
22    /// `aya` rejected the object (BTF, verifier, or format).
23    #[error("eBPF load failed: {0}")]
24    Ebpf(#[from] EbpfError),
25    /// Map conversion failed.
26    #[error("eBPF map error: {0}")]
27    Map(#[from] MapError),
28    /// Required map or program name was absent from the object.
29    #[error("eBPF object missing {0}")]
30    MissingSymbol(&'static str),
31    /// Attach failed (capabilities, kprobe symbol, or permissions).
32    #[error("eBPF attach failed: {0}")]
33    Attach(#[from] ProgramError),
34}
35
36/// High-level observation readiness for status / CLI.
37#[derive(Clone, Copy, Debug, Eq, PartialEq)]
38pub enum ObserverStatus {
39    /// Kprobe attached; ring buffer readable.
40    Attached,
41    /// Load or attach failed; daemon must not claim enforcement.
42    Degraded,
43}
44
45/// Loaded TCP-connect observer. Dropping detaches with the `Ebpf` object.
46pub struct Observer {
47    bpf: Ebpf,
48}
49
50impl Observer {
51    /// Load bytecode from memory and attach to `tcp_v4_connect`.
52    ///
53    /// `bytecode` must be sufficiently aligned for ELF parsing (use
54    /// [`aya::include_bytes_aligned`] or [`Self::load_path_and_attach`]).
55    ///
56    /// # Errors
57    ///
58    /// Returns [`LoadError`] when the object is invalid, symbols are missing, or
59    /// attach fails (missing caps/BTF/kprobe).
60    pub fn load_and_attach(bytecode: &[u8]) -> Result<Self, LoadError> {
61        let bpf = EbpfLoader::new().load(bytecode)?;
62        Self::attach_loaded(bpf)
63    }
64
65    /// Load bytecode from a filesystem path and attach.
66    ///
67    /// # Errors
68    ///
69    /// Returns [`LoadError`] when the file cannot be read or attach fails.
70    pub fn load_path_and_attach(path: impl AsRef<Path>) -> Result<Self, LoadError> {
71        let bpf = EbpfLoader::new().load_file(path.as_ref())?;
72        Self::attach_loaded(bpf)
73    }
74
75    /// Load the release object embedded beside this crate, then attach.
76    ///
77    /// # Errors
78    ///
79    /// Same as [`Self::load_and_attach`].
80    pub fn load_embedded_and_attach() -> Result<Self, LoadError> {
81        Self::load_and_attach(aya::include_bytes_aligned!(concat!(
82            env!("CARGO_MANIFEST_DIR"),
83            "/bpf/interfire-ebpf-programs"
84        )))
85    }
86
87    fn attach_loaded(mut bpf: Ebpf) -> Result<Self, LoadError> {
88        let program: &mut KProbe = bpf
89            .program_mut(PROGRAM_NAME)
90            .ok_or(LoadError::MissingSymbol(PROGRAM_NAME))?
91            .try_into()?;
92        program.load()?;
93        program.attach("tcp_v4_connect", 0)?;
94        let _ = bpf
95            .map(EVENT_MAP)
96            .ok_or(LoadError::MissingSymbol(EVENT_MAP))?;
97        Ok(Self { bpf })
98    }
99
100    #[must_use]
101    pub const fn status(&self) -> ObserverStatus {
102        ObserverStatus::Attached
103    }
104
105    /// Borrow the ring buffer map for event polling.
106    ///
107    /// # Errors
108    ///
109    /// Returns [`LoadError::MissingSymbol`] when the map is absent, or
110    /// [`LoadError::Map`] when the map type is wrong.
111    pub fn ring_buf(&mut self) -> Result<RingBuf<&mut MapData>, LoadError> {
112        let map = self
113            .bpf
114            .map_mut(EVENT_MAP)
115            .ok_or(LoadError::MissingSymbol(EVENT_MAP))?;
116        Ok(RingBuf::try_from(map)?)
117    }
118}

120#[cfg(test)]
121mod tests {
122    use std::error::Error;
123
124    use super::*;
125
126    fn embedded_bytecode() -> &'static [u8] {
127        aya::include_bytes_aligned!(concat!(
128            env!("CARGO_MANIFEST_DIR"),
129            "/bpf/interfire-ebpf-programs"
130        ))
131    }
132
133    #[test]
134    fn embedded_bytecode_is_non_empty() {
135        assert!(!embedded_bytecode().is_empty());
136    }
137
138    #[test]
139    fn embedded_bytecode_is_valid_elf() {
140        let data = embedded_bytecode();
141        object::read::File::parse(data).expect("embedded eBPF object must parse as ELF");
142    }
143
144    #[test]
145    fn load_error_display_covers_variants() {
146        let bytecode = LoadError::Bytecode(io::Error::other("missing"));
147        assert!(bytecode.to_string().contains("unavailable"));
148        assert!(bytecode.source().is_some());
149
150        let missing = LoadError::MissingSymbol("EVENTS");
151        assert!(missing.to_string().contains("EVENTS"));
152        assert!(missing.source().is_none());
153    }
154
155    #[test]
156    fn load_path_missing_file_is_ebpf_error() {
157        let result = Observer::load_path_and_attach("/tmp/interfire-no-such-ebpf-object");
158        assert!(matches!(result, Err(LoadError::Ebpf(_))));
159    }
160
161    #[test]
162    fn load_path_garbage_is_ebpf_error() {
163        let path =
164            std::env::temp_dir().join(format!("interfire-garbage-ebpf-{}", std::process::id()));
165        std::fs::write(&path, b"not-an-elf").unwrap();
166        let result = Observer::load_path_and_attach(&path);
167        let _ = std::fs::remove_file(&path);
168        assert!(matches!(result, Err(LoadError::Ebpf(_))));
169    }
170
171    #[test]
172    fn load_embedded_without_caps_is_explicit_error() {
173        let result = Observer::load_embedded_and_attach();
174        if let Err(error) = result {
175            let message = error.to_string();
176            assert!(
177                message.contains("eBPF") || message.contains("attach") || message.contains("map"),
178                "unexpected error text: {message}"
179            );
180            assert!(error.source().is_some());
181        }
182    }
183
184    #[test]
185    fn load_without_caps_is_explicit_error() {
186        // Non-root CI must not crash; map create / attach fails with a typed error.
187        let result = Observer::load_and_attach(embedded_bytecode());
188        if let Err(error) = result {
189            let message = error.to_string();
190            assert!(
191                message.contains("eBPF") || message.contains("attach") || message.contains("map"),
192                "unexpected error text: {message}"
193            );
194        }
195    }
